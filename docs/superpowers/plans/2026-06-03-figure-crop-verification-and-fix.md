## Task 2: Build re-crop script with improved AI vision prompts

**Files:**
- Create: `book/scripts/figure_recrop.py`

This script replaces `figure_crop_mcp.py` with a better approach. The key improvement: the AI vision prompt explicitly asks for tight bounding boxes and the script validates that crops aren't full-width strips.

- [ ] **Step 1: Write the re-crop script**

```python
#!/usr/bin/env python3
"""Re-crop figures using AI vision MCP with improved prompts.

Improvements over figure_crop_mcp.py:
- Prompts ask for TIGHT bounding boxes around actual figure content
- Validates crops aren't full-width strips (indicates failure)
- Handles multi-figure pages correctly (unique crop per figure)
- Supports re-cropping individual pages or all pages
"""

import asyncio
import json
import os
import sys
from pathlib import Path
from PIL import Image

REPO_ROOT = Path(__file__).resolve().parent.parent.parent
BOOK_DIR = REPO_ROOT / "book"
PAGES_DIR = BOOK_DIR / "pages"
CROP_DIR = PAGES_DIR / "crops"
INVENTORY_PATH = BOOK_DIR / "scripts" / "figure_inventory.json"

PAGE_WIDTH = 1190
PAGE_HEIGHT = 1684


def load_inventory() -> list[dict]:
    """Load figure inventory and return list of figure entries."""
    with open(INVENTORY_PATH) as f:
        return json.load(f)


def group_by_page(figures: list[dict]) -> dict[int, list[dict]]:
    """Group figures by their actual page (from src, not inventory page)."""
    from collections import defaultdict
    groups = defaultdict(list)
    for fig in figures:
        # Use the page number from the src path (e.g., page_020_fig1.png -> 20)
        src_name = fig["src"].split("/")[-1]
        page_num = int(src_name.split("_")[1])
        groups[page_num].append(fig)
    return dict(groups)


def build_vision_prompt(captions: list[str], page_num: int) -> str:
    """Build the AI vision prompt for figure boundary detection."""
    caption_list = "\n".join(f"  {i+1}. {c}" for i, c in enumerate(captions))
    return (
        f"You are analyzing page {page_num} of a Chinese technical book about electricity markets. "
        f"Your task is to find the EXACT pixel boundaries of each standalone figure/diagram/chart.\n\n"
        f"CRITICAL RULES:\n"
        f"- Return bounding boxes that are TIGHT around the figure content, NOT full page width\n"
        f"- Figures are typically centered with margins — the bounding box should NOT span the full 1190px width\n"
        f"- Include the figure caption text as part of the bounding box\n"
        f"- Exclude page numbers, headers, footers, and paragraph text\n"
        f"- The page image is 1190x1684 pixels\n"
        f"- Order figures top-to-bottom, then left-to-right\n\n"
        f"This page has {len(captions)} figure(s) with these captions:\n"
        f"{caption_list}\n\n"
        f"Return ONLY a JSON array. Each object must have:\n"
        f'  "index": figure number (1-based)\n'
        f'  "x": left edge in pixels (should be > 50 for most figures)\n'
        f'  "y": top edge in pixels\n'
        f'  "width": width in pixels (should be < 1100 for most figures)\n'
        f'  "height": height in pixels\n'
        f'  "caption": the matched caption text\n'
        f'  "confidence": "high" or "low"\n'
    )


def parse_vision_response(text: str) -> list[dict]:
    """Parse the AI vision response into a list of bounding boxes."""
    import re
    text = text.strip()
    if text.startswith("```"):
        text = re.sub(r"^```(?:json)?\s*", "", text)
        text = re.sub(r"\s*```$", "", text)
    match = re.search(r"\[\s*\{.*\}\s*\]", text, flags=re.S)
    if match:
        text = match.group(0)
    data = json.loads(text)
    if isinstance(data, dict):
        data = [data]
    return data


def validate_crop(box: dict, page_width: int = PAGE_WIDTH) -> dict:
    """Validate that a crop bounding box is reasonable."""
    x = box.get("x", 0)
    y = box.get("y", 0)
    w = box.get("width", 0)
    h = box.get("height", 0)

    issues = []
    if w >= page_width - 20:
        issues.append("full_width_strip")
    if w * h < 10000:
        issues.append("too_small")
    if w <= 0 or h <= 0:
        issues.append("invalid_dimensions")
    if x < 0 or y < 0:
        issues.append("negative_coordinates")

    return {
        "valid": len(issues) == 0,
        "issues": issues,
        "box": {"x": x, "y": y, "width": w, "height": h},
    }


def crop_image(page_path: Path, box: dict, dest_path: Path, padding: int = 8) -> bool:
    """Crop a region from the page image with padding."""
    with Image.open(page_path) as img:
        x = max(0, box["x"] - padding)
        y = max(0, box["y"] - padding)
        right = min(img.width, box["x"] + box["width"] + padding)
        bottom = min(img.height, box["y"] + box["height"] + padding)
        cropped = img.crop((x, y, right, bottom))
        dest_path.parent.mkdir(parents=True, exist_ok=True)
        cropped.save(dest_path)
        return True


async def recrop_page(
    page_num: int,
    figures: list[dict],
    force: bool = False,
) -> list[dict]:
    """Re-crop all figures on a single page using AI vision MCP.

    Returns list of results: {"figure": ..., "status": "ok"|"skipped"|"failed", "reason": ...}
    """
    results = []
    page_path = PAGES_DIR / f"page_{page_num:03d}.png"

    if not page_path.exists():
        for fig in figures:
            results.append({"figure": fig, "status": "failed", "reason": f"page image missing: {page_path}"})
        return results

    captions = [fig.get("caption") or fig.get("alt", "") for fig in figures]
    prompt = build_vision_prompt(captions, page_num)

    # NOTE: The actual MCP call will be done interactively in Claude Code
    # This script provides the logic; the MCP calls happen in the plan execution
    print(f"PAGE {page_num}: Need to analyze with prompt:\n{prompt}\n")

    return results


def main():
    """Main entry point for batch re-cropping."""
    import argparse
    parser = argparse.ArgumentParser(description="Re-crop figures with AI vision")
    parser.add_argument("--page", type=int, help="Re-crop a specific page")
    parser.add_argument("--all", action="store_true", help="Re-crop all pages")
    parser.add_argument("--check", action="store_true", help="Only check, don't crop")
    parser.add_argument("--force", action="store_true", help="Force re-crop even if crop exists")
    args = parser.parse_args()

    inventory = load_inventory()
    by_page = group_by_page(inventory)

    if args.page:
        page = args.page
        figs = by_page.get(page, [])
        if not figs:
            print(f"No figures found on page {page}")
            sys.exit(1)
        print(f"Page {page}: {len(figs)} figure(s)")
        for fig in figs:
            print(f"  fig {fig['fig_number']}: {fig['caption'][:60]}")

        # Check existing crops
        for fig in figs:
            crop_path = CROP_DIR / fig["expected_crop_name"]
            if crop_path.exists():
                img = Image.open(crop_path)
                w, h = img.size
                is_strip = w >= PAGE_WIDTH - 20
                status = "STRIP (likely broken)" if is_strip else "OK"
                print(f"  Existing crop: {crop_path.name} = {w}x{h} [{status}]")
            else:
                print(f"  No existing crop: {crop_path.name}")
    elif args.check:
        print(f"Checking {len(by_page)} pages with figures...")
        broken = 0
        ok = 0
        for page_num in sorted(by_page.keys()):
            figs = by_page[page_num]
            for fig in figs:
                crop_path = CROP_DIR / fig["expected_crop_name"]
                if crop_path.exists():
                    img = Image.open(crop_path)
                    w, h = img.size
                    if w >= PAGE_WIDTH - 20:
                        broken += 1
                    else:
                        ok += 1
                else:
                    broken += 1
        print(f"Results: {ok} OK, {broken} broken/missing out of {len(inventory)} total")
    else:
        print(f"Found {len(inventory)} figures across {len(by_page)} pages")
        print("Use --page N to process a specific page")
        print("Use --check to audit all existing crops")
        print("Use --all to re-crop everything (interactive)")


if __name__ == "__main__":
    main()
```
